# training/train_emotion.py
import os, librosa, numpy as np, torch, torch.nn as nn
from sklearn.model_selection import train_test_split
from torch.utils.data import Dataset, DataLoader

EMOTIONS = {
    '01': 'neutral', '02': 'calm',    '03': 'happy',
    '04': 'sad',     '05': 'angry',   '06': 'fearful',
    '07': 'disgust', '08': 'surprised'
}
USE_EMOTIONS = ['neutral', 'happy', 'sad', 'angry', 'fearful', 'surprised']
EMOTION2IDX = {e: i for i, e in enumerate(USE_EMOTIONS)}

def extract_features(path):
    y, sr = librosa.load(path, sr=22050, duration=3.0)
    if len(y) < sr * 0.5:
        return None
    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=40)
    pitch, _ = librosa.piptrack(y=y, sr=sr)
    energy = librosa.feature.rms(y=y)
    zcr = librosa.feature.zero_crossing_rate(y)
    spec = librosa.feature.spectral_centroid(y=y, sr=sr)
    feats = np.concatenate([
        mfcc.mean(axis=1), mfcc.std(axis=1),
        [pitch[pitch > 0].mean() if pitch[pitch > 0].size > 0 else 0],
        [energy.mean()], [zcr.mean()], [spec.mean()]
    ])
    return feats.astype(np.float32)   # 84-dim

class EmotionModel(nn.Module):
    def __init__(self, input_dim=84, num_classes=6):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, 256), nn.BatchNorm1d(256),
            nn.ReLU(), nn.Dropout(0.4),
            nn.Linear(256, 128), nn.ReLU(), nn.Dropout(0.3),
            nn.Linear(128, 64), nn.ReLU(),
            nn.Linear(64, num_classes)
        )
    def forward(self, x): return self.net(x)

Step 2.2 — NMT Model Architecture (Day 3)

# models/nmt/seq2seq.py
import torch, torch.nn as nn, torch.nn.functional as F

class Encoder(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, n_layers, dropout):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.rnn = nn.LSTM(embed_dim, hidden_dim, n_layers,
                           bidirectional=True, dropout=dropout, batch_first=True)
        self.fc_h = nn.Linear(hidden_dim * 2, hidden_dim)
        self.fc_c = nn.Linear(hidden_dim * 2, hidden_dim)

    def forward(self, src):
        embedded = self.embedding(src)
        outputs, (h, c) = self.rnn(embedded)
        h = torch.tanh(self.fc_h(torch.cat([h[-2], h[-1]], dim=1)))
        c = torch.tanh(self.fc_c(torch.cat([c[-2], c[-1]], dim=1)))
        return outputs, h.unsqueeze(0), c.unsqueeze(0)

class Attention(nn.Module):
    def __init__(self, enc_dim, dec_dim):
        super().__init__()
        self.attn = nn.Linear(enc_dim * 2 + dec_dim, dec_dim)
        self.v = nn.Linear(dec_dim, 1, bias=False)

    def forward(self, dec_hidden, enc_outputs):
        src_len = enc_outputs.shape[1]
        dec_hidden = dec_hidden.squeeze(0).unsqueeze(1).repeat(1, src_len, 1)
        energy = torch.tanh(self.attn(torch.cat([dec_hidden, enc_outputs], dim=2)))
        return F.softmax(self.v(energy).squeeze(2), dim=1)

class Decoder(nn.Module):
    def __init__(self, vocab_size, embed_dim, enc_dim, dec_dim, n_layers, dropout):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.attention = Attention(enc_dim, dec_dim)
        self.rnn = nn.LSTM(embed_dim + enc_dim * 2, dec_dim, n_layers,
                           dropout=dropout, batch_first=True)
        self.fc_out = nn.Linear(dec_dim + enc_dim * 2 + embed_dim, vocab_size)
        self.dropout = nn.Dropout(dropout)

    def forward(self, tgt_token, dec_h, dec_c, enc_outputs):
        embedded = self.dropout(self.embedding(tgt_token.unsqueeze(1)))
        attn_w = self.attention(dec_h, enc_outputs)
        context = torch.bmm(attn_w.unsqueeze(1), enc_outputs)
        rnn_input = torch.cat([embedded, context], dim=2)
        output, (dec_h, dec_c) = self.rnn(rnn_input, (dec_h, dec_c))
        prediction = self.fc_out(torch.cat([output, context, embedded], dim=2).squeeze(1))
        return prediction, dec_h, dec_c, attn_w

from torch.optim.lr_scheduler import ReduceLROnPlateau

def train_epoch(model, loader, optimizer, criterion, clip=1.0):
    model.train()
    epoch_loss = 0
    for src, tgt in loader:
        optimizer.zero_grad()
        output = model(src, tgt, teacher_forcing_ratio=0.5)
        output = output[:, 1:].reshape(-1, output.shape[-1])
        tgt = tgt[:, 1:].reshape(-1)
        loss = criterion(output, tgt)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), clip)
        optimizer.step()
        epoch_loss += loss.item()
    return epoch_loss / len(loader)

Step 2.4 — Beam Search Decoding (Day 5)

# Better than greedy search — try multiple paths, pick best
def beam_search_translate(model, sentence, src_vocab, tgt_vocab,
                           max_len=50, beam_width=5, device='cpu'):
    model.eval()
    tokens = tokenize_english(sentence)
    src = torch.tensor([src_vocab.get(t, src_vocab['']) for t in tokens],
                       dtype=torch.long).unsqueeze(0).to(device)
    with torch.no_grad():
        enc_out, h, c = model.encoder(src)

    beams = [(0.0, [tgt_vocab['']], h, c)]
    completed = []

    for _ in range(max_len):
        new_beams = []
        for score, tokens_so_far, h, c in beams:
            last_tok = torch.tensor([tokens_so_far[-1]]).to(device)
            pred, h, c, _ = model.decoder(last_tok, h, c, enc_out)
            log_probs = torch.log_softmax(pred, dim=-1)
            top_k = log_probs.topk(beam_width)
            for prob, tok in zip(top_k.values[0], top_k.indices[0]):
                new_score = score + prob.item()
                new_tokens = tokens_so_far + [tok.item()]
                if tok.item() == tgt_vocab['']:
                    completed.append((new_score / len(new_tokens), new_tokens))
                else:
                    new_beams.append((new_score, new_tokens, h, c))
        beams = sorted(new_beams, key=lambda x: x[0], reverse=True)[:beam_width]

    best = sorted(completed, key=lambda x: x[0], reverse=True)[0][1]
    return ' '.join([tgt_vocab_inv[t] for t in best[1:-1]])

# pipeline/transcriber.py
import whisper

class Transcriber:
    def __init__(self, model_size='medium'):
        self.model = whisper.load_model(model_size)

    def transcribe(self, audio_path):
        result = self.model.transcribe(
            audio_path,
            word_timestamps=True,
            language='en',
            task='transcribe'
        )
        segments = []
        for seg in result['segments']:
            segments.append({
                'text': seg['text'].strip(),
                'start': round(seg['start'], 3),
                'end': round(seg['end'], 3),
                'duration': round(seg['end'] - seg['start'], 3)
            })
        return segments

Step 3.2 — Isochronic Timing

# pipeline/isochronic.py
def estimate_hindi_duration(hindi_text, chars_per_second=8.0):
    """Estimate TTS output duration from character count"""
    chars = len(hindi_text.replace(' ', ''))
    return max(0.3, chars / chars_per_second)

def compute_speed_factor(original_duration, hindi_text):
    """How fast should TTS speak to match original timing?"""
    estimated = estimate_hindi_duration(hindi_text)
    factor = estimated / max(original_duration, 0.1)
    return max(0.7, min(1.6, factor))  # clamp: 0.7x slow → 1.6x fast

# pipeline/video_assembler.py
import ffmpeg
from pydub import AudioSegment
import os

def assemble_dubbed_video(original_video, dubbed_segments, output_path, subtitle_path=None):
    # Get video duration
    probe = ffmpeg.probe(original_video)
    duration_ms = int(float(probe['format']['duration']) * 1000)

    # Build full-length audio track
    final_audio = AudioSegment.silent(duration=duration_ms)
    for seg in dubbed_segments:
        hindi_audio = AudioSegment.from_wav(seg['audio_path'])
        pos_ms = int(seg['start'] * 1000)
        final_audio = final_audio.overlay(hindi_audio, position=pos_ms)

    mixed_audio_path = output_path.replace('.mp4', '_audio.wav')
    final_audio.export(mixed_audio_path, format='wav')

    # Merge audio with video
    video_stream = ffmpeg.input(original_video).video
    audio_stream = ffmpeg.input(mixed_audio_path).audio

    if subtitle_path:
        out = ffmpeg.output(
            video_stream, audio_stream, output_path,
            vf=f"subtitles={subtitle_path}",
            acodec='aac', vcodec='libx264', shortest=None
        )
    else:
        out = ffmpeg.output(video_stream, audio_stream, output_path,
                            acodec='aac', vcodec='copy', shortest=None)
    out.run(overwrite_output=True, quiet=True)
    os.remove(mixed_audio_path)
    return output_path

# pipeline/input_handler.py
import yt_dlp, os, uuid, shutil
from pathlib import Path

UPLOAD_DIR = Path("uploads")
UPLOAD_DIR.mkdir(exist_ok=True)

def download_youtube(url: str) -> str:
    """Download YouTube video, return local file path"""
    job_id = str(uuid.uuid4())[:8]
    output_path = str(UPLOAD_DIR / f"{job_id}.mp4")

    ydl_opts = {
        'format': 'bestvideo[height<=720][ext=mp4]+bestaudio[ext=m4a]/best[height<=720]',
        'outtmpl': output_path,
        'merge_output_format': 'mp4',
        'quiet': True,
        'no_warnings': True,
        # Limit: max 10 minutes for demo purposes
        'match_filter': yt_dlp.utils.match_filter_func("duration < 600"),
    }
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        ydl.download([url])

    return output_path

def save_uploaded_file(file_bytes: bytes, filename: str) -> str:
    """Save uploaded video file, return local path"""
    ext = Path(filename).suffix.lower()
    if ext not in ['.mp4', '.mkv', '.avi', '.mov', '.webm']:
        raise ValueError(f"Unsupported format: {ext}")
    job_id = str(uuid.uuid4())[:8]
    output_path = str(UPLOAD_DIR / f"{job_id}{ext}")
    with open(output_path, 'wb') as f:
        f.write(file_bytes)
    return output_path

def is_youtube_url(url: str) -> bool:
    return any(domain in url for domain in ['youtube.com', 'youtu.be'])

# api/database.py
import sqlite3, json
from datetime import datetime

DB_PATH = "cinesync_jobs.db"

def init_db():
    with sqlite3.connect(DB_PATH) as conn:
        conn.execute('''
            CREATE TABLE IF NOT EXISTS jobs (
                job_id TEXT PRIMARY KEY,
                status TEXT,            -- queued|downloading|transcribing|translating|synthesizing|assembling|complete|failed
                source TEXT,
                output_path TEXT,
                progress INTEGER DEFAULT 0,
                error TEXT,
                created_at TEXT,
                updated_at TEXT,
                metadata TEXT           -- JSON: duration, num_segments, etc.
            )
        ''')

def update_job(job_id, status, **kwargs):
    with sqlite3.connect(DB_PATH) as conn:
        existing = conn.execute('SELECT * FROM jobs WHERE job_id=?', (job_id,)).fetchone()
        now = datetime.utcnow().isoformat()
        if existing is None:
            conn.execute(
                'INSERT INTO jobs (job_id,status,source,output_path,progress,error,created_at,updated_at,metadata) VALUES (?,?,?,?,?,?,?,?,?)',
                (job_id, status, kwargs.get('source',''), None, 0, None, now, now, '{}')
            )
        else:
            fields, values = [], []
            for k, v in {'status':status, **kwargs}.items():
                if k in ('status','output_path','progress','error','metadata'):
                    fields.append(f"{k}=?")
                    values.append(json.dumps(v) if k=='metadata' else v)
            fields.append("updated_at=?"); values.append(now)
            values.append(job_id)
            conn.execute(f"UPDATE jobs SET {', '.join(fields)} WHERE job_id=?", values)

def get_job(job_id):
    with sqlite3.connect(DB_PATH) as conn:
        conn.row_factory = sqlite3.Row
        row = conn.execute('SELECT * FROM jobs WHERE job_id=?', (job_id,)).fetchone()
        return dict(row) if row else None

Step 5.3 — Background Worker

# api/worker.py
from pipeline.input_handler import download_youtube
from pipeline.transcriber import Transcriber
from pipeline.emotion_detector import EmotionDetector
from pipeline.translator import Translator
from pipeline.isochronic import compute_speed_factor
from pipeline.tts_synthesizer import HindiSynthesizer
from pipeline.video_assembler import assemble_dubbed_video
from api.database import update_job
import ffmpeg, os
from pathlib import Path

transcriber  = Transcriber(model_size='medium')
emotion_det  = EmotionDetector()
translator   = Translator()
synthesizer  = HindiSynthesizer()

OUTPUTS_DIR = Path("outputs")
OUTPUTS_DIR.mkdir(exist_ok=True)

def run_dubbing_pipeline(job_id: str, source: str, is_youtube: bool):
    try:
        # 1. Download if YouTube
        if is_youtube:
            update_job(job_id, 'downloading', progress=5)
            video_path = download_youtube(source)
        else:
            video_path = source

        # 2. Extract audio
        update_job(job_id, 'extracting_audio', progress=10)
        audio_path = video_path.replace('.mp4', '_audio.wav')
        ffmpeg.input(video_path).output(audio_path, ar=16000, ac=1).run(overwrite_output=True, quiet=True)

        # 3. Transcribe
        update_job(job_id, 'transcribing', progress=20)
        segments = transcriber.transcribe(audio_path)

        # 4. Translate + synthesize each segment
        dubbed_segments = []
        n = len(segments)
        for i, seg in enumerate(segments):
            progress = 25 + int((i / n) * 55)
            update_job(job_id, 'translating', progress=progress)

            hindi_text = translator.translate(seg['text'])
            emotion = emotion_det.detect(audio_path, seg['start'], seg['end'])
            speed = compute_speed_factor(seg['duration'], hindi_text)

            audio_out = str(OUTPUTS_DIR / f"{job_id}_seg_{i:04d}.wav")
            synthesizer.synthesize(hindi_text, emotion, audio_out, speed=speed)
            dubbed_segments.append({'start': seg['start'], 'audio_path': audio_out,
                                     'text': hindi_text, 'emotion': emotion})

        # 5. Assemble video
        update_job(job_id, 'assembling', progress=85)
        output_path = str(OUTPUTS_DIR / f"{job_id}_dubbed.mp4")
        assemble_dubbed_video(video_path, dubbed_segments, output_path)

        # 6. Cleanup temp WAVs
        for seg in dubbed_segments:
            os.remove(seg['audio_path'])

        update_job(job_id, 'complete', output_path=output_path, progress=100)

    except Exception as e:
        update_job(job_id, 'failed', error=str(e))

Interview Prep

❓
"Walk me through your architecture"
"5 stages: Whisper extracts transcript with timestamps, emotion CNN classifies each segment's tone from MFCC features, BiLSTM+Attention model translates to Hindi, Coqui XTTS synthesizes with matching emotion, FFmpeg assembles the final video."
❓
"What is isochronic translation?"
"Matching dubbed audio duration to original. I estimate TTS duration from character count (~8 chars/sec Hindi), compute a speed factor, and clamp it between 0.7x and 1.6x. Without this, audio overruns scene cuts."
❓
"Why BiLSTM over Transformer?"
"To show I understand attention mechanism from scratch — not just call HuggingFace. I also benchmarked against Helsinki-NLP/opus-mt-en-hi. My BLEU was lower (22 vs 31) but I understood every line of training code."
❓
"How did you handle async jobs?"
"FastAPI BackgroundTasks + SQLite job tracking. Client gets job_id immediately, polls /status. Dubbing a 5-min video takes 3-8 minutes — synchronous HTTP would timeout."
❓
"What is your BLEU score?"
"22.4 on IIT Bombay test set. BLEU above 20 is considered reasonable for NMT. Professional quality requires 30+, which I benchmarked against using Helsinki pretrained model."
❓
"What would you improve?"
"1) Replace BiLSTM with fine-tuned mBART for 30+ BLEU. 2) Add Wav2Lip for lip sync. 3) Voice cloning — clone the original speaker's Hindi voice, not just style. 4) Real-time streaming instead of batch."
